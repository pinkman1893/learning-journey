# STL 精简版源码学习

通过精简实现理解 `vector`、`list` 和 `deque` 的内部结构、迭代器设计与常用操作。

## vector
```cpp
#include <iostream>
#include <algorithm>

using namespace std;

template <class T>
class vector {
public:
    // --- 【核心变化】内置 Iterator 类 ---
    class iterator {
        // 让 vector 类能访问 iterator 的私有成员（比如 ptr）
        friend class vector<T>;

    private:
        T* ptr; // 迭代器内部依然是握着一个指针

    public:
        // 构造函数
        iterator(T* p = nullptr) : ptr(p) {}

        // 1. 重载解引用运算符 *
        // 让 *it 就像 *ptr 一样工作
        T& operator*() { return *ptr; }

        // 2. 重载前置 ++ (例如 ++it)
        iterator& operator++() {
            ptr++;
            return *this;
        }

        // 3. 重载后置 ++ (例如 it++)
        iterator operator++(int) {
            iterator temp = *this;
            ptr++;
            return temp;
        }

        // 4. 重载 -- (例如 --it)
        iterator& operator--() {
            ptr--;
            return *this;
        }

        // 5. 重载比较运算符 == 和 !=
        bool operator==(const iterator& other) const { return ptr == other.ptr; }
        bool operator!=(const iterator& other) const { return ptr != other.ptr; }

        // 6. 支持随机访问 (it + n)
        iterator operator+(int n) { return iterator(ptr + n); }
        iterator operator-(int n) { return iterator(ptr - n); }

        // 7. 计算两个迭代器的距离 (it1 - it2)
        int operator-(const iterator& other) const { return ptr - other.ptr; }
    };

private:
    T* start;           // 数组头
    T* finish;          // 数组有效尾
    T* end_of_storage;  // 数组容量尾

    // --- 扩容逻辑 (不变) ---
    void expand_capacity() {
        unsigned int old_size = size();
        unsigned int old_cap = capacity();
        unsigned int new_cap = (old_cap == 0) ? 1 : old_cap * 2;

        T* new_start = new T[new_cap];

        for (unsigned int i = 0; i < old_size; ++i) {
            new_start[i] = start[i];
        }

        if (start != nullptr) delete[] start;

        start = new_start;
        finish = start + old_size;
        end_of_storage = start + new_cap;
    }

public:
    // --- 构造与析构 ---
    vector() : start(nullptr), finish(nullptr), end_of_storage(nullptr) {}

    vector(unsigned int n) {
        start = new T[n];
        finish = start + n;
        end_of_storage = start + n;
        // 初始化
        for (unsigned int i = 0; i < n; ++i) start[i] = T();
    }

    vector(const vector<T>& x) {
        unsigned int n = x.size();
        start = new T[n];
        finish = start + n;
        end_of_storage = start + n;
        for (unsigned int i = 0; i < n; ++i) start[i] = x[i];
    }

    ~vector() {
        if (start != nullptr) delete[] start;
    }

    // --- 赋值重载 ---
    vector<T>& operator=(const vector<T>& other) {
        if (this == &other) return *this;
        if (start != nullptr) delete[] start;

        unsigned int n = other.size();
        start = new T[n];
        finish = start + n;
        end_of_storage = start + n;
        for (unsigned int i = 0; i < n; ++i) start[i] = other[i];
        return *this;
    }

    // --- 核心操作 ---

    void push_back(const T& x) {
        if (finish == end_of_storage) expand_capacity();
        *finish = x;
        finish++;
    }

    // 【重点】参数现在是 iterator 类
    iterator insert(iterator pos, const T& x) {
        // 利用重载的 operator- 计算下标
        int offset = pos - begin();

        if (finish == end_of_storage) {
            expand_capacity();
            // 扩容后指针变了，要重新生成迭代器
            // 利用重载的 operator+
            pos = begin() + offset;
        }

        // 移动数据：注意这里要通过 pos.ptr 访问内部指针，或者直接用迭代器操作
        // 为了方便理解，我们直接用迭代器操作
        for (iterator it = end(); it != pos; --it) {
             // *(it) = *(it - 1) 的逻辑
             // 但因为 end() 指向最后元素的下一位，我们要小心操作
             // 这里为了代码简单，我们用回指针操作
             // 通过 friend 我们可以访问 pos.ptr
        }

        // 简单实现：用指针搬运
        T* p_pos = start + offset;
        for (T* p = finish; p > p_pos; --p) {
            *p = *(p - 1);
        }

        *p_pos = x;
        finish++;

        return begin() + offset;
    }

    void pop_back() {
        if (finish > start) finish--;
    }

    iterator erase(iterator pos) {
        // 计算偏移量
        int offset = pos - begin();
        T* p_pos = start + offset;

        if (p_pos >= start && p_pos < finish) {
            for (T* p = p_pos; p < finish - 1; ++p) {
                *p = *(p + 1);
            }
            finish--;
        }
        return begin() + offset;
    }

    // 区间删除
    void erase(iterator first, iterator last) {
        int n = last - first;
        if (n <= 0) return;

        T* dest = start + (first - begin());
        T* src = start + (last - begin());

        while (src < finish) {
            *dest = *src;
            dest++;
            src++;
        }
        finish -= n;
    }

    // --- 状态与访问 ---
    unsigned int size() const { return finish - start; }
    unsigned int capacity() const { return end_of_storage - start; }
    bool empty() const { return start == finish; }

    T& operator[](unsigned int n) { return start[n]; }
    const T& operator[](unsigned int n) const { return start[n]; }

    T& front() { return *start; }
    T& back() { return *(finish - 1); }

    // --- 生成迭代器 ---
    // 返回一个包装了 start 指针的 iterator 对象
    iterator begin() { return iterator(start); }

    // 返回一个包装了 finish 指针的 iterator 对象
    iterator end() { return iterator(finish); }
};

// --- 测试代码 ---
int main() {
    vector<int> v;
    v.push_back(10);
    v.push_back(20);
    v.push_back(30);

    // 1. 使用内置迭代器类遍历
    // 这里的 it 是一个对象，不是指针，但用起来像指针
    cout << "Traverse: ";
    for (vector<int>::iterator it = v.begin(); it != v.end(); ++it) {
        cout << *it << " "; // 调用了 operator*
    }
    cout << endl;

    // 2. 测试 insert (需要迭代器加减法)
    v.insert(v.begin() + 1, 99);
    // Expect: 10, 99, 20, 30

    cout << "After insert: " << v[1] << endl;

    // 3. 测试 erase
    v.erase(v.begin());
    // Expect: 99, 20, 30

    cout << "Front after erase: " << v.front() << endl;

    return 0;
}
```

## List
```cpp
#include <iostream>
#include <algorithm> // 用于 std::swap
#include <cassert>   // 用于 assert

using namespace std;

template <class T>
class MyList {
protected:
    // --- 1. 内部节点定义 (放在 protected 以便子类扩展) ---
    struct ListNode {
        T data;
        ListNode* prev;
        ListNode* next;

        // 构造函数
        ListNode(const T& val = T())
            : data(val), prev(nullptr), next(nullptr) {}
    };

    // --- 2. 核心数据成员 ---
    ListNode* node;  // 哨兵节点 (End Marker / Dummy Head)
    int _size;       // 链表长度

    // --- 3. 内部辅助函数 ---

    // 初始化空链表 (构建环形结构)
    void init() {
        node = new ListNode(); // 创建哨兵
        node->next = node;     // 自己指自己
        node->prev = node;     // 自己指自己
        _size = 0;
    }

    // 核心搬运逻辑 (将 [first, last) 区间的节点移动到 position 之前)
    // 这是 List 最强大的功能，也是 splice, merge, sort 的基础
    void transfer(ListNode* position, ListNode* first, ListNode* last) {
        if (position != last) {
            // 1. 将 [first, last) 从原链表摘除
            last->prev->next = position;
            first->prev->next = last;
            position->prev->next = first;

            // 2. 重新连接指针
            ListNode* tmp = position->prev;
            position->prev = last->prev;
            last->prev = first->prev;
            first->prev = tmp;
        }
    }

public:
    // --- 4. 迭代器定义 ---
    class iterator {
        friend class MyList<T>; // 允许 List 访问 iterator 的私有成员
    protected:
        ListNode* curr; // 只有这里持有指针
    public:
        iterator(ListNode* x = nullptr) : curr(x) {}

        // 运算符重载
        bool operator==(const iterator& x) const { return curr == x.curr; }
        bool operator!=(const iterator& x) const { return curr != x.curr; }

        T& operator*() const { return curr->data; }
        T* operator->() const { return &(curr->data); }

        // 前置 ++
        iterator& operator++() {
            curr = curr->next;
            return *this;
        }
        // 后置 ++
        iterator operator++(int) {
            iterator tmp = *this;
            ++*this;
            return tmp;
        }
        // 前置 --
        iterator& operator--() {
            curr = curr->prev;
            return *this;
        }
        // 后置 --
        iterator operator--(int) {
            iterator tmp = *this;
            --*this;
            return tmp;
        }
    };

    // --- 5. 构造、析构与赋值 (Rule of Three) ---

    // 默认构造
    MyList() { init(); }

    // 拷贝构造
    MyList(const MyList& x) {
        init();
        for (auto it = x.begin(); it != x.end(); ++it) {
            push_back(*it);
        }
    }

    // 赋值运算符
    MyList& operator=(const MyList& x) {
        if (this != &x) {
            clear(); // 先清空自己
            for (auto it = x.begin(); it != x.end(); ++it) {
                push_back(*it);
            }
        }
        return *this;
    }

    // 析构函数
    ~MyList() {
        clear();     // 删除所有数据节点
        delete node; // 删除哨兵节点
    }

    // --- 6. 常用接口 ---

    iterator begin() const { return iterator(node->next); }
    iterator end() const { return iterator(node); } // 哨兵就是 end

    bool empty() const { return _size == 0; }
    int size() const { return _size; }

    T& front() { return node->next->data; }
    T& back() { return node->prev->data; }

    // --- 7. 核心操作 (Insert & Erase) ---

    // 在 position 之前插入 x
    iterator insert(iterator position, const T& x) {
        ListNode* tmp = new ListNode(x);
        ListNode* p = position.curr; // 当前位置节点

        // 经典的四步连线
        tmp->next = p;
        tmp->prev = p->prev;
        p->prev->next = tmp;
        p->prev = tmp;

        _size++;
        return iterator(tmp);
    }

    // 删除 position 处的节点
    iterator erase(iterator position) {
        ListNode* p = position.curr;
        iterator ret(p->next); // 保存返回值

        // 重新连线，架空 p
        p->prev->next = p->next;
        p->next->prev = p->prev;

        delete p;
        _size--;
        return ret;
    }

    // --- 8. 便捷操作 (复用 Insert/Erase) ---

    void push_back(const T& x) { insert(end(), x); }
    void push_front(const T& x) { insert(begin(), x); }

    void pop_back() { erase(--end()); }
    void pop_front() { erase(begin()); }

    void clear() {
        ListNode* cur = node->next;
        while (cur != node) { // 只要没转回到哨兵
            ListNode* tmp = cur;
            cur = cur->next;
            delete tmp;
        }
        // 恢复初始状态
        node->next = node;
        node->prev = node;
        _size = 0;
    }
};

// --- 测试主程序 ---
int main() {
    MyList<int> myList;

    // 测试 Push
    myList.push_back(10);
    myList.push_back(20);
    myList.push_front(5);
    // List: 5 -> 10 -> 20

    // 测试 Insert
    auto it = myList.begin();
    it++; // 指向 10
    myList.insert(it, 8);
    // List: 5 -> 8 -> 10 -> 20

    // 测试遍历
    cout << "List Content: ";
    for (auto i = myList.begin(); i != myList.end(); ++i) {
        cout << *i << " ";
    }
    cout << endl;

    // 测试 Copy Constructor
    MyList<int> copyList = myList;
    copyList.pop_back(); // 删掉 20

    cout << "Original Size: " << myList.size() << endl; // 4
    cout << "Copy Size: " << copyList.size() << endl;   // 3

    return 0;
}
```

## Deque

> 以下代码用于理解 `deque` 的分段缓冲区与中控器结构，是精简示意代码，并非可直接编译的完整实现。

```cpp
#ifndef DEQUE_H
#define DEQUE_H

// 假设这些基础库依然存在，仅作引用
#include <function.h>
#include <algobase.h>
#include <iterator.h>
#include <bool.h>
#include <memory> // 使用标准库的allocator作为示意

template <class T>
class deque {
public:
    // ---------------------------------------------------------
    // 1. 迭代器设计 (Deque最复杂的部分)
    // ---------------------------------------------------------
    // Deque 的迭代器必须能够“跨越”不同的缓冲区
    class iterator : public random_access_iterator<T, ptrdiff_t> {
        friend class deque<T>;

    protected:
        T* cur;       // 指向当前缓冲区中的当前元素
        T* first;     // 指向当前缓冲区的头部
        T* last;      // 指向当前缓冲区的尾部（含边界）
        T** node;     // 指向中控器(Map)中指向当前缓冲区的那个指针

        // 构造函数
        iterator(T* x, T** y, size_t buffer_size)
            : cur(x), first(*y), last(*y + buffer_size), node(y) {}

    public:
        // 默认构造
        iterator() : cur(0), first(0), last(0), node(0) {}

        // 解引用，直接获取当前指针的值
        T& operator*() const { return *cur; }

        // 计算两个迭代器之间的距离 (重点考点)
        // 逻辑：(两个缓冲区在Map中的距离 * 缓冲区大小) + (当前元素在各自缓冲区内的偏移量之差)
        ptrdiff_t operator-(const iterator& x) const {
            // 这里假设 buffer_size 是外部获取的，为了简化代码逻辑，保持原意
            size_t buffer_sz = last - first;
            return node == x.node
                ? cur - x.cur
                : ptrdiff_t(buffer_sz * (node - x.node - 1) + (cur - first) + (x.last - x.cur));
        }

        // 前置 ++ (移动到下一个元素)
        iterator& operator++() {
            ++cur;              // 普通指针后移
            if (cur == last) {  // 如果到达了当前缓冲区的边缘
                node++;         // 跳到 Map 的下一个节点
                first = *node;  // 更新 first 为新缓冲区的头
                cur = first;    // 当前指向新缓冲区的第一个元素
                last = first + (last - first); // 更新 last (保持 buffer_size)
            }
            return *this;
        }

        // 后置 ++
        iterator operator++(int)  {
            iterator tmp = *this;
            ++*this;
            return tmp;
        }

        // 前置 -- (移动到上一个元素)
        iterator& operator--() {
            if (cur == first) { // 如果在当前缓冲区的头部
                node--;         // 跳到 Map 的前一个节点
                first = *node;
                last = first + (last - first); // 恢复 buffer_size
                cur = last;     // cur 指向新缓冲区的尾部
            }
            --cur; // 再回退一格 (注意：因为 last 是左闭右开区间，所以先跳buffer设为last，再减减变有效元素)
            return *this;
        }

        // 后置 --
        iterator operator--(int) {
            iterator tmp = *this;
            --*this;
            return tmp;
        }

        // 随机访问 += n (支持随机访问是 Deque 的特性)
        iterator& operator+=(ptrdiff_t n) {
            size_t buffer_sz = last - first;
            ptrdiff_t offset = n + (cur - first); // 计算目标位置相对于当前缓冲区起点的总偏移

            if (offset >= 0 && offset < ptrdiff_t(buffer_sz)) {
                // 目标还在当前缓冲区内
                cur += n;
            } else {
                // 目标跨越了缓冲区，需要计算跨越了多少个节点(node)
                ptrdiff_t node_offset = offset > 0
                    ? offset / ptrdiff_t(buffer_sz)
                    : -ptrdiff_t((-offset + buffer_sz - 1) / buffer_sz);

                // 切换到正确的缓冲区节点
                node += node_offset;
                first = *node;
                last = first + buffer_sz;
                // 定位到新缓冲区内的正确位置
                cur = first + (offset - node_offset * ptrdiff_t(buffer_sz));
            }
            return *this;
        }

        iterator& operator-=(ptrdiff_t n) { return *this += -n; }

        iterator operator+(ptrdiff_t n) const {
            iterator tmp = *this;
            return tmp += n;
        }

        iterator operator-(ptrdiff_t n) const {
            iterator tmp = *this;
            return tmp -= n;
        }

        // 随机存取 operator[]
        T& operator[](ptrdiff_t n) { return *(*this + n); }

        bool operator==(const iterator& x) const { return cur == x.cur; }
        // 比较逻辑：如果在同一个缓冲区比 cur，否则比 node (Map中的位置)
        bool operator<(const iterator& x) const {
            return (node == x.node) ? (cur < x.cur) : (node < x.node);
        }
    };

    // ---------------------------------------------------------
    // 2. Deque 的数据成员
    // ---------------------------------------------------------
protected:
    iterator start;     // 指向第一个缓冲区的第一个元素
    iterator finish;    // 指向最后一个缓冲区的最后一个元素的*下一个位置*

    T** map;            // 中控器：本质是一个指针数组，每个元素指向一个缓冲区
    size_t map_size;    // map 数组的大小（也就是能容纳多少个缓冲区指针）
    size_t length;      // 元素总数 (HP STL 特有维护方式，SGI STL通常通过迭代器相减计算)

    // 内存分配器相关 (模拟)
    static size_t buffer_size; // 每个缓冲区的大小 (通常是 512 字节 / sizeof(T))

    // 辅助函数：负责内存扩容和缩容
    void allocate_at_begin();
    void allocate_at_end();
    void deallocate_at_begin();
    void deallocate_at_end();

public:
    // 构造函数
    deque() : start(), finish(), length(0), map(0), map_size(0) {
        buffer_size = 512 / sizeof(T); // 简化的 buffer size 计算
        if(buffer_size < 8) buffer_size = 8;
        // 初始化时不分配实际空间，直到插入元素
    }

    // 常用接口
    iterator begin() { return start; }
    iterator end() { return finish; }

    bool empty() const { return length == 0; }
    size_t size() const { return length; }

    T& operator[](size_t n) { return *(begin() + n); } // 复用迭代器的 + 操作

    T& front() { return *begin(); }
    T& back() { return *(end() - 1); }

    // ---------------------------------------------------------
    // 3. 核心操作：Push (头插与尾插)
    // ---------------------------------------------------------

    void push_front(const T& x) {
        if (empty() || start.cur == start.first) {
            // 如果为空，或者第一个缓冲区已经满了（cur 指向了 first），需要分配新缓冲区
            allocate_at_begin();
        }
        --start.cur; // 向前移动指针
        // construct(start.cur, x); // 调用构造函数，此处直接赋值模拟
        *start.cur = x;
        ++length;

        // 这是一个修正边界的检查，原始代码逻辑
        // if (end().cur == end().last) allocate_at_end();
    }

    void push_back(const T& x) {
        if (empty()) allocate_at_end(); // 如果还没初始化，先分配

        // construct(finish.cur, x);
        *finish.cur = x;
        ++finish.cur; // 尾指针指向的是“下一个”位置，所以先赋值再++
        ++length;

        // 如果尾指针到达了当前缓冲区的边缘，预先分配下一个缓冲区
        if (finish.cur == finish.last) allocate_at_end();
    }

    // ---------------------------------------------------------
    // 4. 核心操作：Pop (头删与尾删)
    // ---------------------------------------------------------

    void pop_front() {
        // destroy(start.cur); // 析构对象
        ++start.cur; // 逻辑删除，指针后移
        --length;
        // 如果当前缓冲区空了（cur 移到了 last），释放这个缓冲区
        if (empty() || start.cur == start.last)
            deallocate_at_begin();
    }

    void pop_back() {
        // 如果 finish 在当前缓冲区的头部，说明这个缓冲区没数据了，先释放并回退到上一个缓冲区
        if (finish.cur == finish.first) deallocate_at_end();
        --finish.cur;
        // destroy(finish.cur);
        --length;
        if (empty()) deallocate_at_end();
    }

    // ---------------------------------------------------------
    // 5. 核心操作：Insert (插入)
    // ---------------------------------------------------------
    // Deque 的 insert 有个极其重要的优化策略：
    // 判断插入点离头部近还是离尾部近，移动元素较少的一端。
    iterator insert(iterator position, const T& x) {
        if (position == begin()) {
            push_front(x);
            return begin();
        } else if (position == end()) {
            push_back(x);
            return end() - 1;
        } else {
            ptrdiff_t index = position - begin();
            // 核心判断：如果插入点在前半段
            if (index < length / 2) {
                push_front(front()); // 在头部扩展一个空间
                // 将 [begin+1, position] 的元素整体前移
                // copy(...)
                // 简化模拟：手动搬运
                iterator it = begin() + 1; // 新的 begin
                iterator src = begin() + 2; // 原来的 begin+1
                while(it != position) *it++ = *src++;
            } else {
                // 如果插入点在后半段，移动后半部分
                push_back(back());
                // copy_backward(...)
                iterator it = end() - 1;
                iterator src = end() - 2;
                while(it != position) *it-- = *src--;
            }
            // 填入新值
            *position = x;
            return position;
        }
    }

    // Erase 操作同理，也是判断移动哪一边代价更小
    void erase(iterator position) {
        iterator next = position;
        ++next;
        ptrdiff_t index = position - begin();
        if (index < (length / 2)) {
            // 移动前面的元素向后覆盖
            // copy_backward(begin(), position, next);
            pop_front();
        } else {
            // 移动后面的元素向前覆盖
            // copy(next, end(), position);
            pop_back();
        }
    }

    // 析构函数：清空所有元素
    ~deque() { while (!empty()) pop_front(); }
};

            map_size = new_map_size;

        } else {

            *--start.node = p;
            start.first = p;
            start.last = p + buffer_size;
            start.cur = start.last; // start指向新缓冲区的尾部（因为是 push_front）
```
